cmake_minimum_required(VERSION 3.18)
project(echo_pocket_voice LANGUAGES CXX)
find_path(SHERPA_INCLUDE_DIR sherpa-onnx/c-api/c-api.h REQUIRED)
find_library(SHERPA_C_API NAMES sherpa-onnx-c-api REQUIRED)
add_library(echo-pocket STATIC pocket_voice.cpp)
target_compile_features(echo-pocket PUBLIC cxx_std_17)
target_include_directories(echo-pocket PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ${SHERPA_INCLUDE_DIR})
target_link_libraries(echo-pocket PUBLIC ${SHERPA_C_API})
add_executable(echo-pocket-cli pocket_cli.cpp)
target_link_libraries(echo-pocket-cli PRIVATE echo-pocket)
